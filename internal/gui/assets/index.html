MicroForge CLI Reference
mforge --help 
# VM Lifecycle Management:
mforge create --name runner-1 --vcpus 1 --memory 512
mforge create --name static-vm --ip 172.16.0.50 --mac 06:00:ac:10:00:32
mforge run <vm_id>
mforge list
mforge stop <vm_id>
mforge save-state <vm_id>
mforge resume <vm_id>
mforge delete <vm_id>

# SSH Direct Access & Remote Execution:
mforge ssh <vm_id>
mforge ssh <vm_id> -- node -e 'console.log("Hello Node.js")'
mforge ssh <vm_id> -- python3 -c 'print("Hello Python")'
mforge ssh <vm_id> -- sudo whoami

# Network Management:
mforge network status
mforge network list
mforge network inspect <vm_id>
mforge network prune

# Web GUI Server:
mforge gui --port 8080 --addr 0.0.0.0
